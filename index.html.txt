        function salvarMembro(e) {
            e.preventDefault();
            const id = document.getElementById('membroId').value;
            const novo = {
                id: id ? parseInt(id) : Date.now(),
                nome: document.getElementById('membroNome').value,
                apelido: document.getElementById('membroApelido').value,
                cargo: document.getElementById('membroCargo').value,
                status: document.getElementById('membroStatus').value,
                moto: document.getElementById('membroMoto').value,
                telefone: document.getElementById('membroTelefone').value,
                nascimento: document.getElementById('membroNascimento').value,
                mensalidadePaga: false
            };

            if (id) {
                const idx = membros.findIndex(m => m.id === parseInt(id));
                membros[idx] = { ...membros[idx], ...novo };
            } else {
                membros.push(novo);
            }

        function togglePresencaModal(eventoId) {
            const ev = eventos.find(e => e.id === eventoId);
            if (!ev) return;
            
            const membroNome = prompt("Digite o Apelido do Membro para confirmar/remover presença:");
            if (!membroNome) return;

            const mb = membros.find(m => m.apelido.toLowerCase() === membroNome.toLowerCase() || m.nome.toLowerCase().includes(membroNome.toLowerCase()));
            if (mb) {
                if (!ev.confirmados) ev.confirmados = [];
                const idx = ev.confirmados.indexOf(mb.id);
                if (idx > -1) {
                    ev.confirmados.splice(idx, 1);
                    alert(`Presença de ${mb.apelido} removida.`);
                } else {
                    ev.confirmados.push(mb.id);
                    alert(`Presença de ${mb.apelido} confirmada!`);
                }
                saveState();
            } else {
                alert("Integrante não encontrado. Verifique o apelido exato.");
            }
        }

        function toggleMensalidade(membroId) {
            const mb = membros.find(m => m.id === membroId);
            if (mb) {
                mb.mensalidadePaga = !mb.mensalidadePaga;
                if (mb.mensalidadePaga) {
                    // Lançar entrada automática
                    transacoes.push({
                        id: Date.now(),
                        descricao: `Mensalidade: ${mb.apelido}`,
                        tipo: "Entrada",
                        valor: 30.00,
                        data: new Date().toISOString().split('T')[0]
                    });
                }
                saveState();
            }
        }
